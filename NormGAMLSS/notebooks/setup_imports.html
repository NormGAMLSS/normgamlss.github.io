Setup and Imports

Setup and Imports#

# --- General scientific stack ------------------------------------------------
# Used all through the tutorial. Every section also imports what it needs at
# the point it needs it, so you can start reading from any page.
import os
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# --- R integration -----------------------------------------------------------
# rpy2 is the Python-to-R bridge that GAMLSS runs on. A few sections reach into
# the R session directly (e.g. robjects.r[model_name]) or call R packages such
# as base and stats for summary(), AIC() and BIC().
import rpy2.robjects as robjects
from rpy2.robjects.packages import importr

base = importr('base')
stats = importr('stats')

# --- NormGAMLSS package ------------------------------------------------------
# The package is split into a few modules:
#
#   normgamlss.gamlss_main          the Gamlss model class (fit, predict,
#                                      z-scores, centiles, plotting)
#   normgamlss.utils                data preparation and bookkeeping
#   normgamlss.model_utils          model comparison and permutation testing
#   normgamlss.site_transfer_utils  multi-site transfer and sampling
#   normgamlss.plot_style           shared figure style and plot helpers
#
# One example from each is imported below to show where things live. The rest
# are imported in the sections that use them, so each page stands on its own.
from normgamlss.gamlss_main import Gamlss
from normgamlss.utils import check_test_sites, filter_benign_r_warnings
from normgamlss.model_utils import model_comparison
from normgamlss.site_transfer_utils import random_sampling
from normgamlss.plot_style import apply_publication_style

# Silence a few harmless R warnings that otherwise repeat on every fit and
# prediction (empty library paths, GAMLSS's internal contrast= typo, and
# predict.gamlss's "safe predictions" refit notice). Every other R warning,
# and every R error, still comes through untouched.
filter_benign_r_warnings()
