run_multiple_formula_models(create_model_func, y_val_columns, df_train, formulas, 
                            x_vals=['age', 'sex', 'site'], main_folder=None, save_log=True, 
                            num_cores=None, force_overwrite=False, save_models=False, 
                            use_multiprocessing=True, formula_names=None, eval_data=None, 
                            affected=None, func_name=None)

    Compares multiple GAMLSS model formulas across different y-values with optional model evaluation. Supports parallel processing for efficiency and caches results to avoid recomputation.

    Parameters:

        create_model_func : callable
            Function to create a GAMLSS model

        y_val_columns : list of str
            List of y-value column names to model (one outcome may also be given as a string, e.g. 'thickness'). All outcomes and formulas are checked before anything is fitted, and one error lists every problem found

        df_train : pandas.DataFrame
            Training dataframe

        formulas : list of dict
            List of formula dictionaries (each with ‘mu_formula’, ‘sigma_formula’, ‘nu_formula’, ‘tau_formula’)

        x_vals : list of str, default=[‘age’, ‘sex’, ‘site’]
            List of predictor variables

        main_folder : str, default=’results’
            Folder for saving results and logs

        save_log : bool, default=True
            Whether to save logs

        num_cores : int, optional
            Number of CPU cores for multiprocessing (uses 4 if None)

        force_overwrite : bool, default=False
            Whether to overwrite existing cached results. If False, a result saved earlier in main_folder is reused only when it was made with the same settings (formulas, model code, data, number of permutations, evaluation data); results made with other settings are computed again, with a warning. The settings are saved next to each result in a small _settings.json file

        save_models : bool, default=False
            Whether to save fitted models

        use_multiprocessing : bool, default=True
            Whether to use multiprocessing

        formula_names : list of str, optional
            Names for the formulas (defaults to “formula1”, “formula2”, etc.)

        eval_data : pandas.DataFrame, optional
            DataFrame for evaluation

        affected : array-like, optional
            Array indicating affected status for evaluation. If None, assumes all subjects are healthy. It must have one value per row of eval_data (checked before anything is fitted)

        func_name : str, optional
            Distribution family name for evaluation (auto-detected if None)

    Returns:

        comparison_df : pandas.DataFrame
            AIC and BIC values for each formula and y-value

        convergence_df : pandas.DataFrame
            Convergence information for each formula and y-value

        summary_df : pandas.DataFrame
            Best models by AIC and BIC for each y-value

        eval_results : dict, optional
            Evaluation metrics (returned if eval_data provided)

import multiprocessing as mp
import os

from normgamlss.gamlss_main import Gamlss
from normgamlss.model_utils import run_multiple_formula_models

# This function gives us the flexibility on what we want to change and what we want to keep constant
# For example, we can keep the nu and tau formula fixed by removing it from this creat_model_func and in the next cell we can
# just change the other parameters or even change the way we want to input

def create_model_func(model_name, x_vals, y_val, mu_formula, sigma_formula, nu_formula, tau_formula):
    def model_func(data):
        gamlss_model = Gamlss(model_name, x_vals, y_val)
        gamlss_model.fit(
            r_code=f"""
            model <- gamlss(
                y ~ {mu_formula},
                sigma.formula = ~ {sigma_formula},
                nu.formula = ~ {nu_formula},
                tau.formula = ~ {tau_formula},
                family = SHASH(),
                method = mixed(10, 100)
            )
            """,
            data=data
        )
        return gamlss_model
    return model_func

Loading Models from Batch Operations#

When using batch operations like run_multiple_formula_models, model_comparison, or run_models_with_permutation, models are automatically saved in organized directory structures.

Model Naming Convention:

  • Formula comparison: "model_{y_val}_{formula_name}.rds"

  • Base/Target comparison: "model_{y_val}_{model_type}.rds" where model_type is ‘base’ or ‘target’

  • Permutation testing: "model_{y_val}_{model_type}_perm_{seed}.rds" for permuted models

Note: The models are saved in {main_folder}/model_cache/saved_models/

Examples:

## Example code to load a saved model

# Update these values to match your setup
main_folder = r"../../results/formula_comparison"  # Path to your main results folder
y_val = "y_thickness2"  # Replace with your outcome variable
formula_name = "formula1"  # String for the formula number

# The rest is derived automatically from the values above
model_name = f"model_{y_val}_{formula_name}"
model_path = os.path.join(main_folder, "model_cache", "saved_models", f"{model_name}.rds")
loaded_model = Gamlss.load_model(model_path)
